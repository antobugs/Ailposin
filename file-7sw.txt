--============================ fs =============================
local function fsOK()
    return writefile and readfile and isfile and isfolder and makefolder and listfiles and delfile
end
local function ensureDir()
    if not fsOK() then return false end
    if not isfolder("TCOBuilder") then pcall(makefolder, "TCOBuilder") end
    if not isfolder(CFG.saveDir) then pcall(makefolder, CFG.saveDir) end
    return isfolder(CFG.saveDir)
end
local function safeName(n)
    n = tostring(n or "build"):lower():gsub("[^%w_%-]", "_"):sub(1, 32)
    if n == "" then n = "build" end
    return n
end
local function listSaves()
    local out = {}
    if not fsOK() or not isfolder(CFG.saveDir) then return out end
    local ok, files = pcall(listfiles, CFG.saveDir)
    if not ok or type(files) ~= "table" then return out end
    for _, f in ipairs(files) do
        local n = tostring(f):match("([^/\\]+)%.json$")
        if n then table.insert(out, n) end
    end
    table.sort(out)
    return out
end

local function setStatus(t) if statusLbl then statusLbl.Text = "» " .. t end end

local function equip(name)
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = LP.Backpack:FindFirstChild(name) or char:FindFirstChild(name)
    if not tool then return nil end
    if tool.Parent ~= char then pcall(hum.EquipTool, hum, tool); task.wait(0.15) end
    local s = tool:FindFirstChild("Script")
    return s and s:FindFirstChild("Event")
end

local function fire(ev, ...)
    if not ev then return false end
    local ok = pcall(ev.FireServer, ev, ...)
    task.wait(delay)
    return ok
end

local function groundRay(pos, ignoreOwn)
    local base = {}
    if LP.Character then table.insert(base, LP.Character) end
    if ignoreOwn then
        for b in pairs(owned) do if b.Parent then table.insert(base, b) end end
    end
    local l1 = table.clone(base)
    if workspace.Terrain then table.insert(l1, workspace.Terrain) end
    local rp1 = RaycastParams.new()
    rp1.FilterType = Enum.RaycastFilterType.Exclude
    rp1.FilterDescendantsInstances = l1
    local h = workspace:Raycast(pos + Vector3.new(0, 100, 0), Vector3.new(0, -260, 0), rp1)
    if h then return h, false end
    local rp2 = RaycastParams.new()
    rp2.FilterType = Enum.RaycastFilterType.Exclude
    rp2.FilterDescendantsInstances = base
    local h2 = workspace:Raycast(pos + Vector3.new(0, 100, 0), Vector3.new(0, -260, 0), rp2)
    if h2 then return h2, true end
    return nil, false
end

local function snapTo(v, s) return math.floor(v / s + 0.5) * s end

local function findNew(expect, tol)
    local f = workspace:FindFirstChild("Bricks")
    if not f then return nil end
    local best, bd = nil, math.huge
    for _, d in ipairs(f:GetDescendants()) do
        if d:IsA("BasePart") and not owned[d] then
            local dist = (d.Position - expect).Magnitude
            if dist < tol and dist < bd then best, bd = d, dist end
        end
    end
    return best
end

local function waitNew(expect, tol)
    local t0 = os.clock()
    while os.clock() - t0 < CFG.findWait do
        local b = findNew(expect, tol)
        if b then owned[b] = true return b end
        task.wait(0.02)
    end
    return nil
end

local function gate()
    while paused and running do task.wait(0.1) end
    return running
end

local function foreignSet()

            end
        end
    end
    return s
end

local function httpPost(url, body, headers)
    local fn = request or http_request or (syn and syn.request) or (fluxus and fluxus.request)
    if not fn then return nil, "executor has no http" end
    local ok, res = pcall(fn, { Url = url, Method = "POST", Headers = headers, Body = body })
    if not ok or type(res) ~= "table" then return nil, "request failed" end
    return res
end

local function extractJSON(s)
    s = tostring(s or ""):gsub("^%s*```[%w]*%s*", ""):gsub("%s*```%s*$", "")
    local a, b = s:find("{"), s:match(".*()}")
    if not a or not b or b < a then return nil end
    return s:sub(a, b)
end

local function askAI(prompt, interior)
    local saves = listSaves()
    local idx = #saves > 0 and ("Saved builds: " .. table.concat(saves, ", ")) or ""
    local mode = interior
        and "INTERIOR: also add furniture inside (beds, tables, chairs, shelves, lamps, carpets) as caps at low heights."
        or "No furniture."
    local msg = SYSTEM .. "\n" .. idx .. "\n" .. mode .. "\n\nBUILD: " .. prompt

    local body = HttpService:JSONEncode({
        model = CFG.model, temperature = 0.5, max_completion_tokens = 3000, top_p = 0.95,
        include_reasoning = false, reasoning_effort = "low",
        response_format = { type = "json_object" },
        messages = { { role = "user", content = msg } },
    })
    local res, err = httpPost("https://api.groq.com/openai/v1/chat/completions", body, {
        ["Content-Type"] = "application/json", ["Authorization"] = "Bearer " .. CFG.apiKey,
    })
    if not res then return nil, err end
    if res.StatusCode ~= 200 then return nil, "HTTP " .. tostring(res.StatusCode) .. " " .. tostring(res.Body):sub(1, 120) end
    local ok, outer = pcall(HttpService.JSONDecode, HttpService, res.Body)
    if not ok or type(outer) ~= "table" or not outer.choices then return nil, "bad envelope" end
    local j = extractJSON(outer.choices[1] and outer.choices[1].message and outer.choices[1].message.content)
    if not j then return nil, "no json" end
    local ok2, plan = pcall(HttpService.JSONDecode, HttpService, j)
    if not ok2 or type(plan) ~= "table" then return nil, "json broken" end
    return plan
end

--============================ plan =============================
local function toColor(hex, fb)
    if type(hex) == "string" then
        local h = hex:gsub("#", "")
        if #h == 6 then
            local ok, c = pcall(function()
                return Color3.fromRGB(tonumber(h:sub(1,2),16), tonumber(h:sub(3,4),16), tonumber(h:sub(5,6),16))
            end)
            if ok and c then return c end
        end
    end
    return fb or Color3.fromRGB(200, 195, 185)
end

local function cleanPlan(raw)
    local plan = { walls = {}, caps = {} }
    local dc, dm = toColor(raw.color), matOf(raw.material)
    local wraw, craw = raw.walls or {}, raw.caps or {}

    local span = 1
    for _, w in ipairs(wraw) do
        span = math.max(span, math.abs(tonumber(w.x) or 0), math.abs(tonumber(w.z) or 0))
    end
    local scale = span > CFG.maxSpan and CFG.maxSpan / span or 1

    local seen = {}
    for _, w in ipairs(wraw) do
        local x, z, h = tonumber(w.x), tonumber(w.z), tonumber(w.h)
        if x and z and h then
            x, z = math.floor(x * scale + 0.5), math.floor(z * scale + 0.5)
            h = math.clamp(math.floor(h), 1, CFG.maxH)
            local k = x .. ":" .. z
            if not seen[k] then
                seen[k] = true
                table.insert(plan.walls, { x = x, z = z, h = h,
                    color = toColor(w.color, dc), mat = matOf(w.material or dm) })
            end
        end
    end
    for _, c in ipairs(craw) do
        local x, y, z = tonumber(c.x), tonumber(c.y), tonumber(c.z)
        if x and y and z then
            table.insert(plan.caps, {
                x = math.floor(x * scale + 0.5), z = math.floor(z * scale + 0.5),
                y = math.clamp(math.floor(y), 1, CFG.maxH + 2),
                color = toColor(c.color, dc), mat = matOf(c.material or dm) })
        end
    end
    plan.name = tostring(raw.name or "build")
    plan.count = #plan.walls + #plan.caps
    return plan
end

local function serialize(plan, prompt)
    local w, c = {}, {}
    for _, v in ipairs(plan.walls) do
        table.insert(w, { x = v.x, z = v.z, h = v.h,
            color = string.format("#%02x%02x%02x", v.color.R*255, v.color.G*255, v.color.B*255), material = v.mat })
    end
    for _, v in ipairs(plan.caps) do
        table.insert(c, { x = v.x, y = v.y, z = v.z,
            color = string.format("#%02x%02x%02x", v.color.R*255, v.color.G*255, v.color.B*255), material = v.mat })
    end
    return HttpService:JSONEncode({ name = plan.name, prompt = prompt or "", walls = w, caps = c })
end

--============================ builder =============================
local anchor, gridOff, total, done, failed

local function world(x, z)
    return anchor + Vector3.new(x * S_CELL, 0, z * S_CELL) + (gridOff or Vector3.zero)
end

local function tick()
    done += 1
    progFill.Size = UDim2.fromScale(math.min(done / total, 1), 1)
    setStatus(math.floor(done / total * 100) .. "%   " .. done .. "/" .. total .. "   fail " .. failed)
end

-- place one cell stacked on `below` (nil = ground)
local function placeCell(x, z, y, below, evBuild)
    local want = world(x, z) + Vector3.new(0, (y - 0.5) * S_CELL, 0)
    local part, face, pos

    if below and below.Parent then
        part, face, pos = below, Enum.NormalId.Top, below.Position + Vector3.new(0, below.Size.Y / 2, 0)
    else
        local g = groundRay(want, true)
        if not g then return nil end
        part, face, pos = g.Instance, Enum.NormalId.Top, g.Position
    end

    for _ = 1, 2 do
        pcall(fire, evBuild, part, face, pos, S_CELL >= 4 and "normal" or "detailed")
        local b = waitNew(want, S_CELL * 1.6)
        if b then
            if not gridOff then
                local d = b.Position - Vector3.new(want.X, b.Position.Y, want.Z)
                gridOff = Vector3.new(d.X, 0, d.Z)
            end
            return b
        end
    end
    return nil
end

local function build(plan, prompt)
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return setStatus("no character") end
    if not workspace:FindFirstChild("Bricks") then return setStatus("place one brick first") end

    total = plan.count
    if total == 0 then return setStatus("empty plan") end

    local probe = hrp.Position + hrp.CFrame.LookVector * (S_CELL * 2)
    local g, onTerrain = groundRay(probe, true)
    if not g then return setStatus("no ground in front") end
    anchor = Vector3.new(snapTo(g.Position.X, S_CELL), g.Position.Y, snapTo(g.Position.Z, S_CELL))

    infoLbl.Text = ("%s  %d blk  %s"):format(g.Instance.Name:sub(1, 14), total,
        onTerrain and "TERRAIN!" or "ok")

    table.clear(placed); table.clear(owned); table.clear(paintMap)
    gridOff, done, failed = nil, 0, 0
    running, paused = true, false
    btnBuild.Text, btnStop.Visible, btnStop.Text = "working", true, "stop"

    local evBuild = equip("Build")
    if not evBuild then running = false; btnBuild.Text = "build"; btnStop.Visible = false
        return setStatus("Build tool missing") end

    -- walls: solid stack per column
    for _, w in ipairs(plan.walls) do
        if not gate() then break end
        local below
        local okCol = true
        for y = 1, w.h do
            if not gate() then okCol = false break end
            local b = placeCell(w.x, w.z, y, below, evBuild)
            if b then
                below = b
                paintMap[b] = { w.color, w.mat }
                if y == w.h then placed[w.x .. ":" .. w.z] = b end
            else
                failed += 1
                okCol = false
                break
            end
        end
        if not okCol then failed += 1 end
        tick()
    end

    -- caps
    if gate() then
        local sides = {
            { f = Enum.NormalId.Right, o = Vector3.new(1, 0, 0) },
            { f = Enum.NormalId.Left,  o = Vector3.new(-1, 0, 0) },
            { f = Enum.NormalId.Back,  o = Vector3.new(0, 0, 1) },
            { f = Enum.NormalId.Front, o = Vector3.new(0, 0, -1) },
        }
        table.sort(plan.caps, function(a, b)
            if a.y ~= b.y then return a.y < b.y end
            if a.z ~= b.z then return a.z < b.z end
            return a.x < b.x
        end)
        for _, c in ipairs(plan.caps) do
            if not gate() then break end
            local below = placed[c.x .. ":" .. c.z]
            local b
            if below and below.Parent and math.abs(below.Position.Y + below.Size.Y / 2 - (anchor.Y + (c.y - 0.5) * S_CELL)) < S_CELL * 1.6 then
                b = placeCell(c.x, c.z, c.y, below, evBuild)
            else
                for _, s in ipairs(sides) do
                    local nb = placed[(c.x + math.floor(s.o.X)) .. ":" .. (c.z + math.floor(s.o.Z))]
                    if nb and nb.Parent then
                        local want = world(c.x, c.z) + Vector3.new(0, (c.y - 0.5) * S_CELL, 0)
                        pcall(fire, evBuild, nb, s.f, nb.Position + s.o * (nb.Size.X / 2),
                            S_CELL >= 4 and "normal" or "detailed")
                        b = waitNew(want, S_CELL * 1.6)
                        if b then break end
                    end
                end
            end
            if b then
                placed[c.x .. ":" .. c.z] = b
                paintMap[b] = { c.color, c.mat }
            else failed += 1 end
            tick()
        end
    end

    -- paint
    if gate() then
        local evPaint = equip("Paint")
        if evPaint then
            for b, pm in pairs(paintMap) do
                if not gate() then break end
                if b.Parent then
                    pcall(fire, evPaint, b, Enum.NormalId.Front, b.Position, "both", pm[1], pm[2], "")
                end
            end
        else
            setStatus("Paint tool missing")
        end
    end

    running, paused = false, false
    btnBuild.Text, btnStop.Visible = "build", false
    setStatus("done  " .. (total - failed) .. "/" .. total .. "  fail " .. failed)
end

local function undoAll()
    local evDel = equip("Delete")
    local l = {}
    for b in pairs(owned) do if b.Parent then table.insert(l, b) end end
    if #l == 0 then return setStatus("nothing to undo") end
    if not evDel then return setStatus("Delete tool missing") end
    setStatus("removing " .. #l)
    for i = #l, 1, -1 do pcall(fire, evDel, l[i], Enum.NormalId.Front, l[i].Position) end
    table.clear(placed); table.clear(owned); table.clear(paintMap)
    progFill.Size = UDim2.fromScale(0, 1)
    setStatus("cleared")
end

local function savePlan()
    if not lastPlan then return setStatus("nothing to save") end
    if not ensureDir() then return setStatus("no fs") end
    local name = safeName(lastPlan.name)
    local i = 2
    while isfile(CFG.saveDir .. "/" .. name .. ".json") do name = safeName(lastPlan.name) .. "_" .. i; i += 1 end
    if pcall(writefile, CFG.saveDir .. "/" .. name .. ".json", serialize(lastPlan, lastPrompt)) then
        setStatus("saved " .. name); refreshSaves()
    else setStatus("save failed") end
end

-- drag
local dragging, dragStart, startPos
header.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging, dragStart, startPos = true, i.Position, main.Position
    end
end)
UIS.InputChanged:Connect(function(i)
    if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
UIS.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

local INTERIOR_WORDS = { "inside", "interior", "furnish", "furniture", "inner", "fill it",
    "bed", "table", "chair", "shelf", "chest", "lamp", "carpet", "stairs", "kitchen", "decorate" }

btnBuild.MouseButton1Click:Connect(function()
    if running then return end
    local prompt = box.Text:gsub("^%s+", ""):gsub("%s+$", "")
    if prompt == "" then return setStatus("type a prompt") end

    local wi = interiorOn
    local low = prompt:lower()
    for _, k in ipairs(INTERIOR_WORDS) do if low:find(k, 1, true) then wi = true break end end

    logLbl.Text = "thinking..."
    setStatus("asking model")
    task.spawn(function()
        local plan, err = askAI(prompt, wi)
        if not plan then setStatus("ai error"); logLbl.Text = tostring(err) return end
        local p = cleanPlan(plan)
        if p.count < 4 then
            setStatus("plan too small")
            logLbl.Text = HttpService:JSONEncode(plan):sub(1, 200)
            return
        end
        lastPlan, lastPrompt = p, prompt
        logLbl.Text = p.name .. (wi and " [inside]" or "")
        build(p, prompt)
    end)
end)
