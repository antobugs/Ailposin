local function fsOK() return writefile and readfile and isfile and isfolder and makefolder and listfiles and delfile end
local function ensureDir()
    if not fsOK() then return false end
    if not isfolder("TCOBuilder") then pcall(makefolder, "TCOBuilder") end
    if not isfolder(CFG.saveDir) then pcall(makefolder, CFG.saveDir) end
    return isfolder(CFG.saveDir)
end
local function safeName(n)
    n = tostring(n or "build"):lower():gsub("[^%w_%-]", "_"):sub(1, 32)
    if n == "" then n = "build" end
    return n
end
local function listSaves()
    local out = {}
    if not fsOK() or not isfolder(CFG.saveDir) then return out end
    local ok, f = pcall(listfiles, CFG.saveDir)
    if not ok or type(f) ~= "table" then return out end
    for _, v in ipairs(f) do
        local n = tostring(v):match("([^/\\]+)%.json$")
        if n then table.insert(out, n) end
    end
    table.sort(out)
    return out
end

local function groundRay(pos, ignoreOwn)
    local base = {}
    if LP.Character then table.insert(base, LP.Character) end
    if ignoreOwn then for b in pairs(owned) do if b.Parent then table.insert(base, b) end end end
    local l1 = table.clone(base)
    if workspace.Terrain then table.insert(l1, workspace.Terrain) end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = l1
    return workspace:Raycast(pos + Vector3.new(0, 60, 0), Vector3.new(0, -200, 0), rp)
end

local function surfaceUnder()

local function snapTo(v, s) return math.floor(v / s + 0.5) * s end

local function partSnap()
    local s, fast = {}, false
    local f = workspace:FindFirstChild("Bricks")
    if f then
        fast = true
        for _, d in ipairs(f:GetDescendants()) do if d:IsA("BasePart") then s[d] = true end end
    else
        for _, d in ipairs(workspace:GetDescendants()) do if d:IsA("BasePart") then s[d] = true end end
    end
    return s, fast
end

local function scanNew(before, fast, expect)
    local pool
    if fast then
        local f = workspace:FindFirstChild("Bricks")
        if not f then return nil end
        pool = f:GetDescendants()
    else
        pool = workspace:GetDescendants()
    end
    local best, bd = nil, math.huge
    for _, d in ipairs(pool) do
        if d:IsA("BasePart") and not before[d] and not owned[d] then
            local nm = d.Name:lower()
            if fast or nm:find("brick") then
                local dist = (d.Position - expect).Magnitude
                if dist < bd then best, bd = d, dist end
            end
        end
    end
    if best and bd <= S_CELL * 6 then return best end
    return nil
end

local function waitNew(before, fast, expect, maxT)
    local t0 = os.clock()
    while os.clock() - t0 < (maxT or CFG.findWait) do
        local b = scanNew(before, fast, expect)
        if b then owned[b] = true return b end
        task.wait(0.02)
    end
    return nil
end

local function httpPost(url, body, headers)
    local fn = request or http_request or (syn and syn.request) or (fluxus and fluxus.request)
    if not fn then return nil, "executor has no http" end
    local ok, res = pcall(fn, { Url = url, Method = "POST", Headers = headers, Body = body })
    if not ok or type(res) ~= "table" then return nil, "request failed" end
    return res
end

local function extractJSON(s)
    s = tostring(s or ""):gsub("^%s*```[%w]*%s*", ""):gsub("%s*```%s*$", "")
    local a, b = s:find("{"), s:match(".*()}")
    if not a or not b or b < a then return nil end
    return s:sub(a, b)
end

local function apiCall(messages, jsonMode)
    local body = HttpService:JSONEncode({
        model = CFG.model, temperature = 0.6, max_completion_tokens = 4000, top_p = 0.95,
        include_reasoning = false, reasoning_effort = "low",
        response_format = { type = jsonMode and "json_object" or "text" },
        messages = messages,
    })
    local res, err = httpPost("https://api.groq.com/openai/v1/chat/completions", body, {
        ["Content-Type"] = "application/json", ["Authorization"] = "Bearer " .. CFG.apiKey,
    })
    if not res then return nil, err end
    if res.StatusCode ~= 200 then return nil, "HTTP " .. tostring(res.StatusCode) .. " " .. tostring(res.Body):sub(1, 110) end
    local ok, outer = pcall(HttpService.JSONDecode, HttpService, res.Body)
    if not ok or type(outer) ~= "table" or not outer.choices then return nil, "bad envelope" end
    return outer.choices[1] and outer.choices[1].message and outer.choices[1].message.content, nil
end

local function askBuild(prompt, interior)
    local saves = listSaves()
    local tail = {}
    if #saves > 0 then tail[#tail+1] = "Saved builds: " .. table.concat(saves, ", ") end
    if interior then tail[#tail+1] = "Also furnish the inside." end
    tail[#tail+1] = "Follow this exactly: " .. prompt

    local msg = "BUILD: " .. prompt .. "\n\n" .. SYSTEM .. "\n" .. table.concat(tail, "\n")
    local content, err = apiCall({ { role = "user", content = msg } }, true)
    if not content then return nil, err end
    local j = extractJSON(content)
    if not j then return nil, "no json" end
    local ok, plan = pcall(HttpService.JSONDecode, HttpService, j)
    if not ok or type(plan) ~= "table" then return nil, "json broken" end
    return plan
end

local function planDump(p)
    if not p then return "none" end
    local parts, n = {}, 0
    for _, w in ipairs(p.walls) do
        n += 1
        if n <= 45 then parts[#parts+1] = ("w(%d,%d,%d-%d,%s)"):format(w.x, w.z, w.y0, w.y1, w.mat) end
    end
    for _, c in ipairs(p.caps) do
        n += 1
        if n <= 45 then parts[#parts+1] = ("c(%d,%d,%d,%s)"):format(c.x, c.y, c.z, c.mat) end
    end
    return ("%s: %d walls, %d caps | %s"):format(p.name, #p.walls, #p.caps, table.concat(parts, " "))
end

local function toColor3(t, fb)
    if type(t) == "table" and #t >= 3 then
        return Color3.new(math.clamp(tonumber(t[1]) or 0, 0, 1),
            math.clamp(tonumber(t[2]) or 0, 0, 1), math.clamp(tonumber(t[3]) or 0, 0, 1))
    end
    if type(t) == "string" then
        local h = t:gsub("#", "")
        if #h == 6 then
            local ok, c = pcall(function()
                return Color3.fromRGB(tonumber(h:sub(1,2),16), tonumber(h:sub(3,4),16), tonumber(h:sub(5,6),16))
            end)
            if ok and c then return c end
        end
    end
    return fb or Color3.fromRGB(200, 195, 185)
end

local function cellsOf(p, s)
    if type(p) ~= "table" or type(s) ~= "table" then return {} end
    local out = {}
    local x0 = math.floor((tonumber(p[1]) or 0) - (tonumber(s[1]) or 1) / 2 + 0.5)
    local x1 = math.floor((tonumber(p[1]) or 0) + (tonumber(s[1]) or 1) / 2 - 0.5)
    local y0 = math.floor((tonumber(p[2]) or 0) - (tonumber(s[2]) or 1) / 2 + 0.5)
    local y1 = math.floor((tonumber(p[2]) or 0) + (tonumber(s[2]) or 1) / 2 - 0.5)
    local z0 = math.floor((tonumber(p[3]) or 0) - (tonumber(s[3]) or 1) / 2 + 0.5)
    local z1 = math.floor((tonumber(p[3]) or 0) + (tonumber(s[3]) or 1) / 2 - 0.5)
    if x1 < x0 then x1 = x0 end
    if y1 < y0 then y1 = y0 end
    if z1 < z0 then z1 = z0 end
    for x = x0, x1 do
        for z = z0, z1 do
            out[#out+1] = { x = x, z = z, y0 = y0, y1 = y1 }
        end
    end
    return out
end

local function cleanPlan(raw)
    local plan = { walls = {}, caps = {} }
    local dc = toColor3(raw.color)
    local dm = matOf(raw.material)
    local frame, caps = raw.frame, raw.caps
    local boxes = (type(frame) == "table") or (type(caps) == "table" and caps[1] and caps[1].pos)

    if boxes then
        local colMap = {}
        for _, b in ipairs(frame or {}) do
            local col = toColor3(b.color, dc)
            local mat = matOf(b.material or dm)
            for _, c in ipairs(cellsOf(b.pos, b.size)) do
                local k = c.x .. ":" .. c.z
                local cur = colMap[k]
                if not cur then
                    colMap[k] = { x = c.x, z = c.z, y0 = math.max(c.y0, 1), y1 = math.max(c.y1, 1), color = col, mat = mat }
                else
                    if c.y0 < cur.y0 then cur.y0 = math.max(c.y0, 1) end
                    if c.y1 > cur.y1 then cur.y1 = c.y1 end
                end
            end
        end
        for _, v in pairs(colMap) do
            if v.y1 >= v.y0 then table.insert(plan.walls, v) end
        end
        for _, b in ipairs(caps or {}) do
            local col = toColor3(b.color, dc)
            local mat = matOf(b.material or dm)
            for _, c in ipairs(cellsOf(b.pos, b.size)) do
                for y = c.y0, c.y1 do
                    table.insert(plan.caps, { x = c.x, z = c.z, y = math.max(y, 1), color = col, mat = mat })
                end
            end
        end
    else
        for _, w in ipairs(raw.walls or {}) do
            local x, z, h = tonumber(w.x), tonumber(w.z), tonumber(w.h)
            if x and z and h then
                table.insert(plan.walls, { x = math.floor(x), z = math.floor(z), y0 = 1,
                    y1 = math.clamp(math.floor(h), 1, CFG.maxH),
                    color = toColor3(w.color, dc), mat = matOf(w.material or dm) })
            end
        end
        for _, c in ipairs(raw.caps or {}) do
            local x, y, z = tonumber(c.x), tonumber(c.y), tonumber(c.z)
            if x and y and z then
                table.insert(plan.caps, { x = math.floor(x), z = math.floor(z),
                    y = math.clamp(math.floor(y), 1, CFG.maxH + 2),
                    color = toColor3(c.color, dc), mat = matOf(c.material or dm) })
            end
        end
    end

    local span = 1
    for _, w in ipairs(plan.walls) do span = math.max(span, math.abs(w.x), math.abs(w.z)) end
    if span > CFG.maxSpan then
        local f = CFG.maxSpan / span
        for _, w in ipairs(plan.walls) do w.x = math.floor(w.x * f + 0.5); w.z = math.floor(w.z * f + 0.5) end
        for _, c in ipairs(plan.caps) do c.x = math.floor(c.x * f + 0.5); c.z = math.floor(c.z * f + 0.5) end
    end
    for _, w in ipairs(plan.walls) do w.y1 = math.min(w.y1, CFG.maxH) end
    for _, c in ipairs(plan.caps) do c.y = math.min(c.y, CFG.maxH + 2) end

    plan.name = tostring(raw.name or "build")
    plan.count = #plan.walls + #plan.caps
    return plan
end

local function serialize(plan, prompt)
    local fr, cp = {}, {}
    for _, v in ipairs(plan.walls) do
        table.insert(fr, { pos = { v.x, math.floor((v.y0 + v.y1) / 2), v.z }, size = { 1, v.y1 - v.y0 + 1, 1 },
            material = v.mat, color = { v.color.R, v.color.G, v.color.B } })
    end
    for _, v in ipairs(plan.caps) do
        table.insert(cp, { pos = { v.x, v.y, v.z }, size = { 1, 1, 1 }, material = v.mat,
            color = { v.color.R, v.color.G, v.color.B } })
    end
    return HttpService:JSONEncode({ name = plan.name, prompt = prompt or "", frame = fr, caps = cp })
end

local anchor, gridOff, total, done, failed, streak
local function world(x, z) return anchor + Vector3.new(x * S_CELL, 0, z * S_CELL) + (gridOff or Vector3.zero) end

local function tick(ok)
    done += 1
    progFill.Size = UDim2.fromScale(math.min(done / total, 1), 1)
    if ok then streak = 0 else streak += 1 end
    setStatus(math.floor(math.min(done / total, 1) * 100) .. "%  " .. done .. "/" .. total .. "  fail " .. failed)
end

local function placeCell(x, z, y, below, mode)
    local want = world(x, z) + Vector3.new(0, (y - 0.5) * S_CELL, 0)
    ensureNear(want)
    local part, face, pos
    if below and below.Parent then
        part, face, pos = below, Enum.NormalId.Top, below.Position + Vector3.new(0, below.Size.Y / 2, 0)
    else
        local g = groundRay(want, true)
        if not g then return nil end
        part, face, pos = g.Instance, Enum.NormalId.Top, g.Position
    end
    local function attempt(ev, activate)
        if not ev then return nil end
        local before, fast = partSnap()
        pcall(fire, ev, part, face, pos, mode)
        if activate then
            local t = toolOf("Build")
            if t and t.Activate then pcall(t.Activate, t) end
        end
        return waitNew(before, fast, want, 0.8)
    end
    local b
    if not useEquip then
        b = attempt(getEvent("Build"), false)
        if not b then b = attempt(getEvent("Build"), true) end
    end
    if not b then
        task.wait(0.35)
        local ev2 = equip("Build")
        if ev2 then useEquip = true; b = attempt(ev2, false) end
    end
    if not b then
        task.wait(0.6)
        b = attempt(getEvent("Build") or equip("Build"), false)
    end
    if b and not gridOff then
        local d = b.Position - Vector3.new(want.X, b.Position.Y, want.Z)
        gridOff = Vector3.new(d.X, 0, d.Z)
    end
    return b
end

local function resolveAnchor()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil, "no character" end
    local txt = posBox and posBox.Text or ""
    local px, pz = txt:match("(-?%d+%.?%d*)%s*[, ]%s*(-?%d+%.?%d*)")
    local a
    if px and pz then
        local x, z = tonumber(px), tonumber(pz)
        local g = groundRay(Vector3.new(x, hrp.Position.Y, z), true)
        if not g then return nil, "no ground at " .. x .. "," .. z end
        a = Vector3.new(snapTo(x, S_CELL), g.Position.Y, snapTo(z, S_CELL))
    else
        local g = surfaceUnder()
        if not g then return nil, "no block under you" end
        local ahead = groundRay(g.Position + hrp.CFrame.LookVector * (S_CELL * 2), true)
        local base = ahead and ahead.Position or g.Position
        a = Vector3.new(snapTo(base.X, S_CELL), g.Position.Y, snapTo(base.Z, S_CELL))
    end
    local flat = Vector3.new(a.X, hrp.Position.Y, a.Z)
    if (flat - hrp.Position).Magnitude < S_CELL * 0.7 then
        a = a + hrp.CFrame.LookVector * S_CELL
        a = Vector3.new(snapTo(a.X, S_CELL), a.Y, snapTo(a.Z, S_CELL))
    end
    return a, nil
end

local function build(plan, prompt)
    if not plan or plan.count == 0 then return setStatus("nothing to build") end
    total = plan.count
    local a, err = resolveAnchor()
    if not a then return setStatus(err) end
    anchor = a
    infoLbl.Text = ("x%d z%d · %d blk"):format(anchor.X, anchor.Z, total)
    if total > CFG.maxBlocks then log("big: " .. total) end

    table.clear(placed); table.clear(owned); table.clear(paintMap)
    gridOff, done, failed, streak = nil, 0, 0, 0
    useEquip = false
    running, paused = true, false
    btnBuild.Text, btnStop.Visible, btnStop.Text = "working", true, "stop"
    local mode = S_CELL >= 4 and "normal" or "detailed"

    table.sort(plan.walls, function(p, q)
        if p.z ~= q.z then return p.z < q.z end
        return p.x < q.x
    end)

    for _, w in ipairs(plan.walls) do
        if not gate() then break end
        local below, okCol = nil, true
        for y = w.y0, w.y1 do
            if not gate() then okCol = false break end
            local sup = below
            if not sup and y > 1 then
                local k = placed[w.x .. ":" .. w.z]
                if k and k.Parent then sup = k end
            end
            local b = placeCell(w.x, w.z, y, sup, mode)
            if b then
                below = b
                paintMap[b] = { w.color, w.mat }
                if y == w.y1 then placed[w.x .. ":" .. w.z] = b end
                tick(true)
            else
                failed += 1; okCol = false; tick(false); break
            end
        end
        if not okCol then failed += 1 end
        if safeOn and streak >= CFG.safeStreak then
            paused = true; btnStop.Text = "resume"
            setStatus("safe pause: " .. streak .. " fails")
            break
        end
    end

    if gate() then
        local sides = {
            { f = Enum.NormalId.Right, o = Vector3.new(1, 0, 0) },
            { f = Enum.NormalId.Left,  o = Vector3.new(-1, 0, 0) },
            { f = Enum.NormalId.Back,  o = Vector3.new(0, 0, 1) },
            { f = Enum.NormalId.Front, o = Vector3.new(0, 0, -1) },
        }
        table.sort(plan.caps, function(p, q)
            if p.y ~= q.y then return p.y < q.y end
            if p.z ~= q.z then return p.z < q.z end
            return p.x < q.x
        end)
        for _, c in ipairs(plan.caps) do
            if not gate() then break end
            local below = placed[c.x .. ":" .. c.z]
            local b
            if below and below.Parent then
                b = placeCell(c.x, c.z, c.y, below, mode)
            else
                for _, s in ipairs(sides) do
                    local nb = placed[(c.x + math.floor(s.o.X)) .. ":" .. (c.z + math.floor(s.o.Z))]
                    if nb and nb.Parent then
                        local want = world(c.x, c.z) + Vector3.new(0, (c.y - 0.5) * S_CELL, 0)
                        ensureNear(want)
                        local before, fast = partSnap()
                        local ev = useEquip and equip("Build") or getEvent("Build")
                        pcall(fire, ev, nb, s.f, nb.Position + s.o * (nb.Size.X / 2), mode)
                        b = waitNew(before, fast, want, 0.8)
                        if b then break end
                    end
                end
            end
            if b then placed[c.x .. ":" .. c.z] = b; paintMap[b] = { c.color, c.mat }; tick(true)
            else failed += 1; tick(false) end
            if safeOn and streak >= CFG.safeStreak then
                paused = true; btnStop.Text = "resume"; setStatus("safe pause"); break
            end
        end
    end

    if gate() then
        local evP = getEvent("Paint") or equip("Paint")
        if evP then
            for b, pm in pairs(paintMap) do
                if not gate() then break end
                if b.Parent then
                    pcall(fire, evP, b, Enum.NormalId.Front, b.Position, "both", pm[1], pm[2], sprayId or "")
                end
            end
        else setStatus("Paint missing") end
    end

    if not paused then
        running = false
        btnBuild.Text, btnStop.Visible = "build", false
        setStatus("done " .. math.max(total - failed, 0) .. "/" .. total .. "  fail " .. failed)
    end
end

local function deleteOwned()
    local ev = getEvent("Delete") or equip("Delete")
    local l = {}
    for b in pairs(owned) do if b.Parent then table.insert(l, b) end end
    if #l == 0 then return 0 end
    if not ev then return -1 end
    for i = #l, 1, -1 do
        pcall(ev.FireServer, ev, l[i], Enum.NormalId.Front, l[i].Position)
        if delay > 0 then task.wait(delay) end
    end
    table.clear(placed); table.clear(owned); table.clear(paintMap)
    return #l
end

local function killBuild()
    running, paused = false, false
    btnBuild.Text, btnStop.Visible = "build", false
    progFill.Size = UDim2.fromScale(0, 1)
    task.spawn(function()
        local t0 = os.clock()
        while #owned > 0 and os.clock() - t0 < 8 do task.wait(0.1) end
        setStatus("deleting...")
        local n = deleteOwned()
        setStatus(n == -1 and "Delete missing" or ("killed " .. n))
    end)
end

local function probe()
    local f = workspace:FindFirstChild("Bricks")
    local g = surfaceUnder()
    if not g then return setStatus("no block under you") end
    local part, top = g.Instance, g.Position
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    local out = { "Bricks:" .. (f and ("yes " .. #f:GetChildren()) or "NO"),
        "dist:" .. math.floor((hrp.Position - top).Magnitude) }
    local function attempt(name, ev, pos)
        local before, fast = partSnap()
        local ok = pcall(ev.FireServer, ev, part, Enum.NormalId.Top, pos, "normal")
        task.wait(1.0)
        local b = scanNew(before, fast, top)
        out[#out+1] = name .. (b and "=OK" or (ok and "=no" or "=ERR"))
    end
    local evBp = getEvent("Build")
    out[#out+1] = "bp:" .. (evBp and "yes" or "no")
    if evBp then
        attempt("here", evBp, top)
        attempt("front3", evBp, top + hrp.CFrame.LookVector * 3)
        attempt("far10", evBp, top + hrp.CFrame.LookVector * 10)
        attempt("far20", evBp, top + hrp.CFrame.LookVector * 20)
    end
    local evEq = equip("Build")
    out[#out+1] = "eq:" .. (evEq and "yes" or "no")
    if evEq then attempt("eq-here", evEq, top) end
    log(table.concat(out, " "))
    setStatus("probe done")
end

local function refreshSaves()
    for _, c in ipairs(saveList:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end
    local names = listSaves()
    if #names == 0 then log("no saves") return end
    for i = 1, math.min(#names, 30) do
        local n = names[i]
        local fr = Instance.new("Frame")
        fr.Size = UDim2.new(1, -6, 0, 22)
        fr.BackgroundColor3 = Color3.fromRGB(32, 34, 42)
        fr.BorderSizePixel = 0
        fr.Parent = saveList
        local rc = Instance.new("UICorner"); rc.CornerRadius = UDim.new(0, 5); rc.Parent = fr
        local lb = Instance.new("TextButton")
        lb.Size = UDim2.new(1, -28, 1, 0)
        lb.BackgroundTransparency = 1
        lb.Text = "  " .. n
        lb.TextSize = 11
        lb.Font = Enum.Font.Gotham
        lb.TextXAlignment = Enum.TextXAlignment.Left
        lb.TextColor3 = Color3.fromRGB(205, 208, 220)
        lb.Parent = fr
        lb.MouseButton1Click:Connect(function()
            if running or not fsOK() then return end
            local ok, data = pcall(readfile, CFG.saveDir .. "/" .. n .. ".json")
            if not ok or not data then return end
            local ok2, raw = pcall(HttpService.JSONDecode, HttpService, data)
            if ok2 and type(raw) == "table" then
                local p = cleanPlan(raw)
                lastPlan, lastPrompt = p, raw.prompt
                task.spawn(build, p, raw.prompt)
            end
        end)
        local dl = Instance.new("TextButton")
        dl.Size = UDim2.fromOffset(22, 22)
        dl.Position = UDim2.new(1, -24, 0, 0)
        dl.BackgroundTransparency = 1
        dl.Text = "×"
        dl.TextSize = 13
        dl.Font = Enum.Font.GothamBold
        dl.TextColor3 = Color3.fromRGB(220, 110, 110)
        dl.Parent = fr
        dl.MouseButton1Click:Connect(function()
            if fsOK() then pcall(delfile, CFG.saveDir .. "/" .. n .. ".json"); refreshSaves() end
        end)
    end
    log(#names .. " saved")
end

local function savePlan()
    if not lastPlan then return setStatus("nothing to save") end
    if not ensureDir() then return setStatus("no fs") end
    local name, i = safeName(lastPlan.name), 2
    while isfile(CFG.saveDir .. "/" .. name .. ".json") do name = safeName(lastPlan.name) .. "_" .. i; i += 1 end
    if pcall(writefile, CFG.saveDir .. "/" .. name .. ".json", serialize(lastPlan, lastPrompt)) then
        setStatus("saved " .. name); refreshSaves()
    else setStatus("save failed") end
end

local INTERIOR_WORDS = { "inside", "interior", "furnish", "furniture", "inner", "fill it",
    "bed", "table", "chair", "shelf", "chest", "lamp", "carpet", "stairs", "kitchen", "decorate" }

btnBuild.MouseButton1Click:Connect(function()
    local text = box.Text:gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return setStatus("type something") end

    if chatOn then
        box.Text = ""
        pushChat("me", text)
        setStatus("thinking")
        task.spawn(function()
            local reply, err = askChat(text)
            if not reply then pushChat("ai", "error: " .. tostring(err)); setStatus("ai error") return end
            pushChat("ai", reply)
            local j = extractJSON(reply)
            if j then
                local ok, raw = pcall(HttpService.JSONDecode, HttpService, j)
                if ok and type(raw) == "table" and (raw.frame or raw.walls) then
                    pendingPlan = cleanPlan(raw)
                    lastPlan, lastPrompt = pendingPlan, text
                    setStatus("plan ready " .. pendingPlan.count .. " — build")
                    btnBuild.Text = "build it"
                    chatOn = false
                    chatBox.Visible = false
                    resize()
                    btnChat.Text = "chat"
                    btnChat.BackgroundColor3 = Color3.fromRGB(44, 46, 58)
                    box.PlaceholderText = "what to build..."
                end
            end
        end)
        return
    end

    if running then return end

    if pendingPlan and btnBuild.Text == "build it" then
        local p = pendingPlan
        pendingPlan = nil
        btnBuild.Text = "build"
        return task.spawn(build, p, text)
    end

    local wi = interiorOn
    local low = text:lower()
    for _, k in ipairs(INTERIOR_WORDS) do if low:find(k, 1, true) then wi = true break end end

    log("thinking...")
    setStatus("asking model")
    task.spawn(function()
        local plan, err = askBuild(text, wi)
        if not plan then setStatus("ai error"); log(tostring(err)) return end
        local p = cleanPlan(plan)
        lastPlan, lastPrompt = p, text
        log(p.name .. " · " .. #p.walls .. "w " .. #p.caps .. "c" .. (wi and " [inside]" or ""))
        build(p, text)
    end)
end)
