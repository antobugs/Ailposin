local function fsOK() return writefile and readfile and isfile and isfolder and makefolder and listfiles and delfile end
local function ensureDir()
    if not fsOK() then return false end
    if not isfolder("TCOBuilder") then pcall(makefolder, "TCOBuilder") end
    if not isfolder(CFG.saveDir) then pcall(makefolder, CFG.saveDir) end

local function safeName(n)
    n = tostring(n or "build"):lower():gsub("[^%w_%-]", "_"):sub(1, 32)
    if n == "" then n = "build" end
    return n
end
local function listSaves()
    local out = {}
    if not fsOK() or not isfolder(CFG.saveDir) then return out end
    local ok, f = pcall(listfiles, CFG.saveDir)
    if not ok or type(f) ~= "table" then return out end
    for _, v in ipairs(f) do
        local n = tostring(v):match("([^/\\]+)%.json$")
        if n then table.insert(out, n) end
    end
    table.sort(out)
    return out
end

local function equip(name)
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = toolOf(name)
    if not tool then return nil end
    for _ = 1, 3 do
        if tool.Parent == char then break end
        pcall(hum.EquipTool, hum, tool)
        task.wait(0.2)
        if tool.Parent ~= char then pcall(function() tool.Parent = char end); task.wait(0.15) end
        if tool.Parent ~= char then pcall(hum.UnequipTools, hum); task.wait(0.15) end
    end
    if tool.Parent ~= char then return nil end
    local s = tool:FindFirstChild("Script")
    return s and s:FindFirstChild("Event")
end

local function fire(ev, ...)

    local ok = pcall(ev.FireServer, ev, ...)

end

local function bricksFolder() return workspace:FindFirstChild("Bricks") or workspace end

local function groundRay(pos, ignoreOwn)
    local base = {}
    if LP.Character then table.insert(base, LP.Character) end
    if ignoreOwn then for b in pairs(owned) do if b.Parent then table.insert(base, b) end end end
    local l1 = table.clone(base)
    if workspace.Terrain then table.insert(l1, workspace.Terrain) end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = l1
    return workspace:Raycast(pos + Vector3.new(0, 60, 0), Vector3.new(0, -200, 0), rp)
end

local function surfaceUnder()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local p = hrp.Position
    for _, o in ipairs({ Vector3.zero, Vector3.new(4,0,0), Vector3.new(-4,0,0), Vector3.new(0,0,4),
        Vector3.new(0,0,-4), Vector3.new(8,0,0), Vector3.new(-8,0,0), Vector3.new(0,0,8), Vector3.new(0,0,-8) }) do
        local g = groundRay(p + o, true)
        if g then return g end
    end
    return nil
end

local function snapTo(v, s) return math.floor(v / s + 0.5) * s end

local function findNew(expect, tol)
    local best, bd = nil, math.huge
    for _, d in ipairs(bricksFolder():GetDescendants()) do
        if d:IsA("BasePart") and not owned[d] and d ~= workspace.Terrain then
            local dist = (d.Position - expect).Magnitude
            if dist < tol and dist < bd then best, bd = d, dist end
        end
    end
    return best
end

local function waitNew(expect, tol)
    local t0 = os.clock()
    while os.clock() - t0 < CFG.findWait do
        local b = findNew(expect, tol)
        if b then owned[b] = true return b end
        task.wait(0.02)
    end
    return nil
end

local function gate()
    while paused and running do task.wait(0.1) end
    return running
end

local function httpPost(url, body, headers)
    local fn = request or http_request or (syn and syn.request) or (fluxus and fluxus.request)
    if not fn then return nil, "executor has no http" end
    local ok, res = pcall(fn, { Url = url, Method = "POST", Headers = headers, Body = body })
    if not ok or type(res) ~= "table" then return nil, "request failed" end
    return res
end

local function extractJSON(s)
    s = tostring(s or ""):gsub("^%s*```[%w]*%s*", ""):gsub("%s*```%s*$", "")
    local a, b = s:find("{"), s:match(".*()}")
    if not a or not b or b < a then return nil end
    return s:sub(a, b)
end

local function toColor(hex, fb)
    if type(hex) == "string" then
        local h = hex:gsub("#", "")
        if #h == 6 then
            local ok, c = pcall(function()
                return Color3.fromRGB(tonumber(h:sub(1,2),16), tonumber(h:sub(3,4),16), tonumber(h:sub(5,6),16))
            end)
            if ok and c then return c end
        end
    end
    return fb or Color3.fromRGB(200, 195, 185)
end

local function cleanPlan(raw)
    local plan = { walls = {}, caps = {} }
    local dc, dm = toColor(raw.color), matOf(raw.material)
    local wraw, craw = raw.walls or {}, raw.caps or {}
    local span = 1
    for _, w in ipairs(wraw) do span = math.max(span, math.abs(tonumber(w.x) or 0), math.abs(tonumber(w.z) or 0)) end
    local scale = span > CFG.maxSpan and CFG.maxSpan / span or 1
    local seen = {}
    for _, w in ipairs(wraw) do
        local x, z, h = tonumber(w.x), tonumber(w.z), tonumber(w.h)
        if x and z and h then
            x, z = math.floor(x * scale + 0.5), math.floor(z * scale + 0.5)
            local k = x .. ":" .. z
            if not seen[k] then
                seen[k] = true
                table.insert(plan.walls, { x = x, z = z, h = math.clamp(math.floor(h), 1, CFG.maxH),
                    color = toColor(w.color, dc), mat = matOf(w.material or dm), spray = w.spray })
            end
        end
    end
    for _, c in ipairs(craw) do
        local x, y, z = tonumber(c.x), tonumber(c.y), tonumber(c.z)
        if x and y and z then
            table.insert(plan.caps, { x = math.floor(x * scale + 0.5), z = math.floor(z * scale + 0.5),
                y = math.clamp(math.floor(y), 1, CFG.maxH + 2),
                color = toColor(c.color, dc), mat = matOf(c.material or dm), spray = c.spray })
        end
    end
    plan.name = tostring(raw.name or "build")
    plan.count = #plan.walls + #plan.caps
    return plan
end

local function serialize(plan, prompt)
    local w, c = {}, {}
    for _, v in ipairs(plan.walls) do
        table.insert(w, { x = v.x, z = v.z, h = v.h,
            color = string.format("#%02x%02x%02x", v.color.R*255, v.color.G*255, v.color.B*255), material = v.mat, spray = v.spray })
    end
    for _, v in ipairs(plan.caps) do
        table.insert(c, { x = v.x, y = v.y, z = v.z,
            color = string.format("#%02x%02x%02x", v.color.R*255, v.color.G*255, v.color.B*255), material = v.mat, spray = v.spray })
    end
    return HttpService:JSONEncode({ name = plan.name, prompt = prompt or "", walls = w, caps = c })
end

local anchor, gridOff, total, done, failed, streak
local function world(x, z) return anchor + Vector3.new(x * S_CELL, 0, z * S_CELL) + (gridOff or Vector3.zero) end

local function tick(ok)
    done += 1
    progFill.Size = UDim2.fromScale(math.min(done / total, 1), 1)
    if ok then streak = 0 else streak += 1 end
    setStatus(math.floor(math.min(done / total, 1) * 100) .. "%  " .. done .. "/" .. total .. "  fail " .. failed)
end

local function placeCell(x, z, y, below, ev)
    local want = world(x, z) + Vector3.new(0, (y - 0.5) * S_CELL, 0)
    local part, face, pos
    if below and below.Parent then
        part, face, pos = below, Enum.NormalId.Top, below.Position + Vector3.new(0, below.Size.Y / 2, 0)
    else
        local g = groundRay(want, true)
        if not g then return nil end
        part, face, pos = g.Instance, Enum.NormalId.Top, g.Position
    end
    for _ = 1, 2 do
        pcall(fire, ev, part, face, pos, S_CELL >= 4 and "normal" or "detailed")
        local b = waitNew(want, S_CELL * 1.9)
        if b then
            if not gridOff then
                local d = b.Position - Vector3.new(want.X, b.Position.Y, want.Z)
                gridOff = Vector3.new(d.X, 0, d.Z)
            end
            return b
        end
    end
    return nil
end

local function build(plan, prompt)
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return setStatus("no character") end
    total = plan.count
    if total == 0 then return setStatus("empty plan") end

    local g = surfaceUnder()
    if not g then return setStatus("no block under you") end

    -- anchor 2 cells in front of where you stand, so the player is not inside the cell
    local ahead = g.Position + hrp.CFrame.LookVector * (S_CELL * 2)
    local probe = groundRay(ahead, true)
    local base = probe and probe.Position or g.Position
    anchor = Vector3.new(snapTo(base.X, S_CELL), g.Position.Y, snapTo(base.Z, S_CELL))

    local okTools = {}
    for _, n in ipairs({"Build","Paint","Delete"}) do okTools[n] = toolOf(n) and "ok" or "MISSING" end
    infoLbl.Text = ("%s · %d blk · B:%s P:%s"):format(g.Instance.Name:sub(1, 10), total, okTools.Build, okTools.Paint)

    table.clear(placed); table.clear(owned); table.clear(paintMap)
    gridOff, done, failed, streak = nil, 0, 0, 0
    running, paused = true, false
    btnBuild.Text, btnStop.Visible, btnStop.Text = "working", true, "stop"

    local ev = equip("Build")
    if not ev then
        running = false; btnBuild.Text = "build"; btnStop.Visible = false
        return setStatus("Build will not equip")
    end

    for _, w in ipairs(plan.walls) do
        if not gate() then break end
        local below, okCol = nil, true
        for y = 1, w.h do
            if not gate() then okCol = false break end
            local b = placeCell(w.x, w.z, y, below, ev)
            if b then
                below = b
                paintMap[b] = { w.color, w.mat, w.spray }
                if y == w.h then placed[w.x .. ":" .. w.z] = b end
                tick(true)
            else
                failed += 1; okCol = false
                tick(false)
                break
            end
        end
        if not okCol then failed += 1 end
        if safeOn and streak >= CFG.safeStreak then
            paused = true
            btnStop.Text = "resume"
            setStatus("safe: " .. streak .. " fails, paused")
            break
        end
    end

    if gate() then
        local sides = {
            { f = Enum.NormalId.Right, o = Vector3.new(1, 0, 0) },
            { f = Enum.NormalId.Left,  o = Vector3.new(-1, 0, 0) },
            { f = Enum.NormalId.Back,  o = Vector3.new(0, 0, 1) },
            { f = Enum.NormalId.Front, o = Vector3.new(0, 0, -1) },
        }
        table.sort(plan.caps, function(a, b)
            if a.y ~= b.y then return a.y < b.y end
            if a.z ~= b.z then return a.z < b.z end
            return a.x < b.x
        end)
        for _, c in ipairs(plan.caps) do
            if not gate() then break end
            local below = placed[c.x .. ":" .. c.z]
            local b
            if below and below.Parent then
                b = placeCell(c.x, c.z, c.y, below, ev)
            else
                for _, s in ipairs(sides) do
                    local nb = placed[(c.x + math.floor(s.o.X)) .. ":" .. (c.z + math.floor(s.o.Z))]
                    if nb and nb.Parent then
                        local want = world(c.x, c.z) + Vector3.new(0, (c.y - 0.5) * S_CELL, 0)
                        pcall(fire, ev, nb, s.f, nb.Position + s.o * (nb.Size.X / 2), S_CELL >= 4 and "normal" or "detailed")
                        b = waitNew(want, S_CELL * 1.9)
                        if b then break end
                    end
                end
            end
            if b then placed[c.x .. ":" .. c.z] = b; paintMap[b] = { c.color, c.mat, c.spray }; tick(true)
            else failed += 1; tick(false) end
            if safeOn and streak >= CFG.safeStreak then
                paused = true; btnStop.Text = "resume"
                setStatus("safe: paused, too many fails")
                break
            end
        end
    end

    if gate() then
        local evP = equip("Paint")
        if evP then
            for b, pm in pairs(paintMap) do
                if not gate() then break end
                if b.Parent then
                    local id = pm[3] and tostring(pm[3]) or sprayId
                    pcall(fire, evP, b, Enum.NormalId.Front, b.Position, "both", pm[1], pm[2], id or "")
                end
            end
        else setStatus("Paint will not equip") end
    end

    if not paused then
        running = false
        btnBuild.Text, btnStop.Visible = "build", false
        setStatus("done  " .. (total - failed) .. "/" .. total .. "  fail " .. failed)
    end
end

local function deleteOwned()
    local ev = findEvent("Delete")
    local l = {}
    for b in pairs(owned) do if b.Parent then table.insert(l, b) end end
    if #l == 0 then return 0 end
    if not ev then return -1 end
    for i = #l, 1, -1 do
        pcall(ev.FireServer, ev, l[i], Enum.NormalId.Front, l[i].Position)
        if delay > 0 then task.wait(delay) end
    end
    table.clear(placed); table.clear(owned); table.clear(paintMap)
    return #l
end

local function killBuild()
    running, paused = false, false
    btnBuild.Text, btnStop.Visible = "build", false
    progFill.Size = UDim2.fromScale(0, 1)
    task.spawn(function()
        local t0 = os.clock()
        while #owned > 0 and os.clock() - t0 < 8 do task.wait(0.1) end
        setStatus("deleting...")
        equip("Delete")
        local n = deleteOwned()
        if n == -1 then setStatus("Delete missing") else setStatus("killed " .. n) end
    end)
end

-- probe: try several placement strategies, report which one works
local function probe()
    local g = surfaceUnder()
    if not g then return setStatus("no block under you") end
    local ev = equip("Build")
    if not ev then return setStatus("Build will not equip") end
    local tool = toolOf("Build")
    local part, top = g.Instance, g.Position
    local before = #bricksFolder():GetDescendants()
    local results = {}

    local strategies = {
        { "top-surface", function() pcall(ev.FireServer, ev, part, Enum.NormalId.Top, top, "normal") end },
        { "top-center",  function() pcall(ev.FireServer, ev, part, Enum.NormalId.Top, part.Position + Vector3.new(0, part.Size.Y/2, 0), "normal") end },
        { "ahead-4",     function() pcall(ev.FireServer, ev, part, Enum.NormalId.Top, top + LP.Character.HumanoidRootPart.CFrame.LookVector * 4, "normal") end },
        { "activated",   function()
            pcall(ev.FireServer, ev, part, Enum.NormalId.Top, top, "normal")
            if tool and tool.Activate then pcall(tool.Activate, tool) end
        end },
        { "detailed",    function() pcall(ev.FireServer, ev, part, Enum.NormalId.Top, top, "detailed") end },
    }
    for _, s in ipairs(strategies) do
        local n0 = #bricksFolder():GetDescendants()
        s[2]()
        task.wait(1.0)
        local n1 = #bricksFolder():GetDescendants()
        table.insert(results, s[1] .. (n1 > n0 and "=OK" or "=no"))
    end
    log(table.concat(results, "  "))
    setStatus(#bricksFolder():GetDescendants() > before and "probe placed blocks" or "probe: nothing placed")
end

        end)
    end
    log(#names .. " saved")
end

local function savePlan()
    if not lastPlan then return setStatus("nothing to save") end
    if not ensureDir() then return setStatus("no fs") end
    local name, i = safeName(lastPlan.name), 2
    while isfile(CFG.saveDir .. "/" .. name .. ".json") do name = safeName(lastPlan.name) .. "_" .. i; i += 1 end
    if pcall(writefile, CFG.saveDir .. "/" .. name .. ".json", serialize(lastPlan, lastPrompt)) then
        setStatus("saved " .. name); refreshSaves()
    else setStatus("save failed") end
end

local INTERIOR_WORDS = { "inside", "interior", "furnish", "furniture", "inner", "fill it",
    "bed", "table", "chair", "shelf", "chest", "lamp", "carpet", "stairs", "kitchen", "decorate" }

btnBuild.MouseButton1Click:Connect(function()
    if running then return end
    local prompt = box.Text:gsub("^%s+", ""):gsub("%s+$", "")
    if prompt == "" then return setStatus("type a prompt") end
    local wi = interiorOn
    local low = prompt:lower()
    for _, k in ipairs(INTERIOR_WORDS) do if low:find(k, 1, true) then wi = true break end end
    log("thinking...")
    setStatus("asking model")
    task.spawn(function()
        local plan, err = askAI(prompt, wi)
        if not plan then setStatus("ai error"); log(tostring(err)) return end
        local p = cleanPlan(plan)
        if p.count < 4 then
            setStatus("plan too small")
            log(HttpService:JSONEncode(plan):sub(1, 180))
            return
        end
        lastPlan, lastPrompt = p, prompt
        log(p.name .. " · " .. #p.walls .. "w " .. #p.caps .. "c" .. (wi and " [inside]" or ""))
        build(p, prompt)
    end)
end)
