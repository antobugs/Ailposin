--============================ filesystem =============================
local function fsOK()
    return writefile and readfile and isfile and isfolder and makefolder and listfiles and delfile
end

local function ensureDir()
    if not fsOK() then return false end
    if not isfolder("TCOBuilder") then pcall(makefolder, "TCOBuilder") end
    if not isfolder(CFG.saveDir) then pcall(makefolder, CFG.saveDir) end
    return isfolder(CFG.saveDir)
end

local function safeName(n)
    n = tostring(n or "build"):lower():gsub("[^%w_%-]", "_"):sub(1, 32)
    if n == "" then n = "build" end
    return n
end

local function listSaves()
    local out = {}
    if not fsOK() or not isfolder(CFG.saveDir) then return out end
    local ok, files = pcall(listfiles, CFG.saveDir)
    if not ok or type(files) ~= "table" then return out end
    for _, f in ipairs(files) do
        local n = tostring(f):match("([^/\\]+)%.json$")
        if n then table.insert(out, n) end
    end
    table.sort(out)
    return out
end

local function uniqueName(base)
    local n, i = base, 2
    while isfile(CFG.saveDir .. "/" .. n .. ".json") do
        n = base .. "_" .. i
        i += 1
    end
    return n
end

--============================ helpers =============================
local function setStatus(t) if statusLbl then statusLbl.Text = "» " .. t end end

local function toolEvent(name)
    local c = LP.Character
    local tool = LP.Backpack:FindFirstChild(name) or (c and c:FindFirstChild(name))
    if not tool then return nil end
    local s = tool:FindFirstChild("Script")
    return s and s:FindFirstChild("Event")
end

local function fire(ev, ...)
    if not ev then return false end
    local ok = pcall(ev.FireServer, ev, ...)
    task.wait(CFG.fireDelay)
    return ok
end

local function groundRay(pos, extraIgnore)
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    local list = {}
    if LP.Character then table.insert(list, LP.Character) end
    for b in pairs(owned) do if b.Parent then table.insert(list, b) end end
    if extraIgnore then table.insert(list, extraIgnore) end
    rp.FilterDescendantsInstances = list
    return workspace:Raycast(pos + Vector3.new(0, 90, 0), Vector3.new(0, -220, 0), rp)
end

local function snapTo(v, step) return math.floor(v / step + 0.5) * step end

local function findNewBrick(expect, tol)
    local folder = workspace:FindFirstChild("Bricks")
    if not folder then return nil end
    local best, bestD = nil, math.huge
    for _, d in ipairs(folder:GetDescendants()) do
        if d:IsA("BasePart") and not owned[d] then
            local dist = (d.Position - expect).Magnitude
            if dist < tol and dist < bestD then best, bestD = d, dist end
        end
    end
    return best
end

local function waitBrick(expect, tol)
    local t0 = os.clock()
    while os.clock() - t0 < CFG.findTimeout do
        local b = findNewBrick(expect, tol)
        if b then owned[b] = true return b end
        task.wait(0.03)
    end
    return nil
end

local function gate()
    while paused and running do task.wait(0.1) end
    return running
end

local SYSTEM = [[You are a voxel architect for a block building game. Answer with ONE raw JSON object, no prose, no markdown.

{"name":"short name","color":"#rrggbb","material":"plastic",
 "walls":[{"x":0,"z":0,"h":12,"color":"#rrggbb","material":"plastic"}],

- One unit = one 4x4x4 block. x,z integers -8..8, centre 0. h in blocks 2..9, y in blocks 1..12.
- walls = columns built solid from the ground up. This is the whole structure.

- Be decisive, do not overthink. Return JSON only.]]

local function httpPost(url, body, headers)
    local fn = request or http_request or (syn and syn.request) or (fluxus and fluxus.request)
    if not fn then return nil, "executor has no http function" end
    local ok, res = pcall(fn, { Url = url, Method = "POST", Headers = headers, Body = body })
    if not ok or type(res) ~= "table" then return nil, "request failed" end
    return res
end

local function extractJSON(s)
    s = tostring(s or "")
    s = s:gsub("^%s*```[%w]*%s*", ""):gsub("%s*```%s*$", "")
    local a, b = s:find("{"), s:match(".*()}") and s:match(".*()}")
    if not a or not b then return nil end
    return s:sub(a, b)
end

local function askAI(prompt)
    local saves = listSaves()
    local index = #saves > 0 and ("Saved builds available: " .. table.concat(saves, ", ") .. ". If the user names one, reuse its general form.") or ""
    local userMsg = SYSTEM .. "\n\n" .. index .. "\n\nBUILD: " .. prompt

    local body = HttpService:JSONEncode({
        model = CFG.model,
        temperature = 0.5,
        max_completion_tokens = 3000,
        top_p = 0.95,
        include_reasoning = false,
        reasoning_effort = "low",
        response_format = { type = "json_object" },
        messages = { { role = "user", content = userMsg } },
    })
    local res, err = httpPost("https://api.groq.com/openai/v1/chat/completions", body, {
        ["Content-Type"] = "application/json",
        ["Authorization"] = "Bearer " .. CFG.apiKey,
    })
    if not res then return nil, err end
    if res.StatusCode ~= 200 then return nil, "HTTP " .. tostring(res.StatusCode) .. " " .. tostring(res.Body):sub(1, 140) end

    local ok, outer = pcall(HttpService.JSONDecode, HttpService, res.Body)
    if not ok or type(outer) ~= "table" or not outer.choices then return nil, "bad api envelope" end
    local raw = outer.choices[1] and outer.choices[1].message and outer.choices[1].message.content
    local j = extractJSON(raw)
    if not j then return nil, "no json in reply" end
    local ok2, plan = pcall(HttpService.JSONDecode, HttpService, j)
    if not ok2 or type(plan) ~= "table" then return nil, "json did not parse" end
    return plan
end

--============================ plan =============================
local function toColor(hex, fallback)
    if type(hex) == "string" then
        local h = hex:gsub("#", "")
        if #h == 6 then
            local ok, c = pcall(function()
                return Color3.fromRGB(tonumber(h:sub(1,2),16), tonumber(h:sub(3,4),16), tonumber(h:sub(5,6),16))
            end)
            if ok and c then return c end
        end
    end
    return fallback or Color3.fromRGB(200, 200, 200)
end

local function cleanPlan(raw)
    local plan = { walls = {}, caps = {} }
    local dc = toColor(raw.color)
    local dm = MATS[tostring(raw.material)] and tostring(raw.material) or "plastic"
    local seen = {}
    for _, w in ipairs(raw.walls or {}) do
        local x, z, h = tonumber(w.x), tonumber(w.z), tonumber(w.h)
        if x and z and h and math.abs(x) <= 20 and math.abs(z) <= 20 then
            x, z = math.floor(x), math.floor(z)
            local k = x .. ":" .. z
            if not seen[k] then
                seen[k] = true
                table.insert(plan.walls, { x = x, z = z, h = math.clamp(math.floor(h), 1, 30),
                    color = toColor(w.color, dc),
                    mat = MATS[tostring(w.material)] and tostring(w.material) or dm })
            end
        end
    end
    for _, c in ipairs(raw.caps or {}) do
        local x, y, z = tonumber(c.x), tonumber(c.y), tonumber(c.z)
        if x and y and z and math.abs(x) <= 20 and math.abs(z) <= 20 then
            table.insert(plan.caps, { x = math.floor(x), y = math.clamp(math.floor(y), 1, 40), z = math.floor(z),
                color = toColor(c.color, dc),
                mat = MATS[tostring(c.material)] and tostring(c.material) or dm })
        end
    end
    plan.name = tostring(raw.name or "build")
    plan.count = #plan.walls + #plan.caps
    return plan
end

local function serialize(plan, prompt)
    local walls, caps = {}, {}
    for _, w in ipairs(plan.walls) do
        table.insert(walls, { x = w.x, z = w.z, h = w.h,
            color = string.format("#%02x%02x%02x", w.color.R*255, w.color.G*255, w.color.B*255), material = w.mat })
    end
    for _, c in ipairs(plan.caps) do
        table.insert(caps, { x = c.x, y = c.y, z = c.z,
            color = string.format("#%02x%02x%02x", c.color.R*255, c.color.G*255, c.color.B*255), material = c.mat })
    end
    return HttpService:JSONEncode({
        name = plan.name, prompt = prompt or "", date = os.date and os.date("%Y-%m-%d %H:%M") or "",
        walls = walls, caps = caps,
    })
end

local gridOff

local function cellWorld(anchor, x, z)
    return anchor + Vector3.new(x * S_CELL, 0, z * S_CELL) + (gridOff or Vector3.zero)
end

local function paintBrick(brick, color, mat, evPaint)
    if not evPaint then return end
    pcall(fire, evPaint, brick, Enum.NormalId.Front, brick.Position, "both", color, mat, "")
end

local function growColumn(ev, brick, targetY)
    local grown = 0
    while brick and brick.Parent and brick.Position.Y < targetY - 0.4 and grown < CFG.maxGrow do
        if not gate() then return brick end
        pcall(fire, ev, brick, Enum.NormalId.Top, brick.Position + Vector3.new(0, brick.Size.Y / 2, 0), "increase")
        grown += 1
        task.wait(0.02)
    end
    return brick
end

local function build(plan, prompt)
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return setStatus("no character") end

    local evBuild, evShape, evPaint = toolEvent("Build"), toolEvent("Shape"), toolEvent("Paint")
    if not evBuild then return setStatus("equip the Build tool") end
    if not evShape then return setStatus("equip the Shape tool") end
    if not evPaint then return setStatus("equip the Paint tool") end
    if not workspace:FindFirstChild("Bricks") then return setStatus("no Bricks folder") end

    local hit = groundRay(hrp.Position)
    if not hit then return setStatus("stand on ground") end
    local anchor = hit.Position
    anchor = Vector3.new(snapTo(anchor.X, S_CELL), anchor.Y, snapTo(anchor.Z, S_CELL))

    local total = plan.count
    if total == 0 then return setStatus("plan empty") end
    if total > CFG.maxBlocks then return setStatus("too big: " .. total) end

    gridOff = nil
    running, paused = true, false
    btnBuild.Text, btnStop.Visible, btnStop.Text = "working", true, "stop"
    btnSize.Text = "cell " .. S_CELL

    local done, failed = 0, 0
    local function tick()
        done += 1
        progFill.Size = UDim2.fromScale(math.min(done / total, 1), 1)
        setStatus(done .. "/" .. total .. "  (" .. failed .. " failed)")
    end

    local firstW = plan.walls[1]
    if firstW then
        local expect = anchor + Vector3.new(firstW.x * S_CELL, 0, firstW.z * S_CELL)
        local g = groundRay(expect)
        if g then
            pcall(fire, evBuild, g.Instance, Enum.NormalId.Top, g.Position, S_CELL >= 4 and "normal" or "detailed")
            local b = waitBrick(expect, S_CELL * 1.4)
            if b then
                local d = b.Position - Vector3.new(expect.X, b.Position.Y, expect.Z)
                gridOff = Vector3.new(d.X, 0, d.Z)
                S_CELL = math.max(math.floor(b.Size.X + 0.5), 1)
                b = growColumn(evShape, b, anchor.Y + firstW.h * S_CELL)
                if b then
                    paintBrick(b, firstW.color, firstW.mat, evPaint)
                    placed[firstW.x .. ":" .. firstW.z] = b
                else failed += 1 end
            else failed += 1 end
            tick()
        end
    end

    for i = 2, #plan.walls do
        if not gate() then break end
        local w = plan.walls[i]
        local world = cellWorld(anchor, w.x, w.z)
        local g = groundRay(world)
        if g then
            pcall(fire, evBuild, g.Instance, Enum.NormalId.Top, g.Position, S_CELL >= 4 and "normal" or "detailed")
            local b = waitBrick(world, S_CELL * 1.4)
            if b then
                b = growColumn(evShape, b, anchor.Y + w.h * S_CELL)
                paintBrick(b, w.color, w.mat, evPaint)
                placed[w.x .. ":" .. w.z] = b
            else failed += 1 end
        else failed += 1 end
        tick()
        if done % 6 == 0 then task.wait(0.04) end
    end

    local sides = {
        { f = Enum.NormalId.Right, o = Vector3.new(1, 0, 0) },
        { f = Enum.NormalId.Left,  o = Vector3.new(-1, 0, 0) },
        { f = Enum.NormalId.Back,  o = Vector3.new(0, 0, 1) },
        { f = Enum.NormalId.Front, o = Vector3.new(0, 0, -1) },
    }
    table.sort(plan.caps, function(a, b)
        if a.y ~= b.y then return a.y < b.y end
        if a.z ~= b.z then return a.z < b.z end
        return a.x < b.x
    end)
    for _, c in ipairs(plan.caps) do
        if not gate() then break end
        local world = cellWorld(anchor, c.x, c.z)
        local targetY = anchor.Y + c.y * S_CELL
        local brick
        local below = placed[c.x .. ":" .. c.z]
        if below and below.Parent and math.abs(below.Position.Y + below.Size.Y / 2 - targetY) < S_CELL * 1.5 then
            pcall(fire, evBuild, below, Enum.NormalId.Top,
                below.Position + Vector3.new(0, below.Size.Y / 2, 0), S_CELL >= 4 and "normal" or "detailed")
            brick = waitBrick(world, S_CELL * 1.4)
        else
            for _, s in ipairs(sides) do
                local nb = placed[(c.x + math.floor(s.o.X)) .. ":" .. (c.z + math.floor(s.o.Z))]
                if nb and nb.Parent and math.abs(nb.Position.Y - targetY) < S_CELL then
                    pcall(fire, evBuild, nb, s.f, nb.Position + s.o * (nb.Size.X / 2),
                        S_CELL >= 4 and "normal" or "detailed")
                    brick = waitBrick(world, S_CELL * 1.4)
                    if brick then break end
                end
            end
        end
        if brick then
            paintBrick(brick, c.color, c.mat, evPaint)
            placed[c.x .. ":" .. c.z] = brick
        else failed += 1 end
        tick()
        if done % 6 == 0 then task.wait(0.04) end
    end

    running, paused = false, false
    btnBuild.Text, btnStop.Visible = "build", false
    setStatus("done  " .. (total - failed) .. "/" .. total .. "  failed " .. failed)
end

local function undoAll()
    local evDel = toolEvent("Delete")
    local list = {}
    for b in pairs(owned) do if b.Parent then table.insert(list, b) end end
    if #list == 0 then return setStatus("nothing to undo") end
    if not evDel then return setStatus("equip Delete tool to undo") end
    setStatus("removing " .. #list)
    for i = #list, 1, -1 do
        local b = list[i]
        pcall(fire, evDel, b, Enum.NormalId.Front, b.Position)
        task.wait(0.02)
    end
    table.clear(placed); table.clear(owned)
    progFill.Size = UDim2.fromScale(0, 1)
    setStatus("cleared")
end

local function runPlan(plan, prompt)
    lastPlan, lastPrompt = plan, prompt
    task.spawn(build, plan, prompt)
end

local function refreshSaves()
    if not saveList then return end
    for _, c in ipairs(saveList:GetChildren()) do
        if c:IsA("Frame") then c:Destroy() end
    end
    local names = listSaves()
    if #names == 0 then
        logLbl.Text = fsOK() and "no saves yet" or "filesystem unavailable"
        return
    end
    for i = 1, math.min(#names, 40) do
        local n = names[i]
        local rowF = Instance.new("Frame")
        rowF.Size = UDim2.new(1, -6, 0, 24)
        rowF.BackgroundColor3 = Color3.fromRGB(32, 34, 42)
        rowF.BorderSizePixel = 0
        rowF.Parent = saveList
        local rc = Instance.new("UICorner"); rc.CornerRadius = UDim.new(0, 5); rc.Parent = rowF

        local load = Instance.new("TextButton")
        load.Size = UDim2.new(1, -30, 1, 0)
        load.BackgroundTransparency = 1
        load.Text = "  " .. n
        load.TextSize = 11
        load.Font = Enum.Font.Gotham
        load.TextXAlignment = Enum.TextXAlignment.Left
        load.TextColor3 = Color3.fromRGB(200, 205, 220)
        load.Parent = rowF
        load.MouseButton1Click:Connect(function()
            if running then return setStatus("busy") end
            if not fsOK() then return setStatus("no filesystem") end
            local ok, data = pcall(readfile, CFG.saveDir .. "/" .. n .. ".json")
            if not ok or not data then return setStatus("read failed") end
            local ok2, raw = pcall(HttpService.JSONDecode, HttpService, data)
            if not ok2 or type(raw) ~= "table" then return setStatus("corrupt save") end
            local plan = cleanPlan(raw)
            if plan.count == 0 then return setStatus("empty save") end
            logLbl.Text = "loading " .. n
            runPlan(plan, raw.prompt)
        end)

        local del = Instance.new("TextButton")
        del.Size = UDim2.fromOffset(24, 24)
        del.Position = UDim2.new(1, -26, 0, 0)
        del.BackgroundTransparency = 1
        del.Text = "×"
        del.TextSize = 13
        del.Font = Enum.Font.GothamBold
        del.TextColor3 = Color3.fromRGB(220, 110, 110)
        del.Parent = rowF
        del.MouseButton1Click:Connect(function()
            if not fsOK() then return end
            pcall(delfile, CFG.saveDir .. "/" .. n .. ".json")
            refreshSaves()
        end)
    end
    logLbl.Text = #names .. " saved"
end

local function savePlan()
    if not lastPlan then return setStatus("nothing to save") end
    if not ensureDir() then return setStatus("no filesystem") end
    local base = safeName(lastPlan.name)
    local name = uniqueName(base)
    local ok = pcall(writefile, CFG.saveDir .. "/" .. name .. ".json", serialize(lastPlan, lastPrompt))
    if ok then
        lastSaveName = name
        setStatus("saved as " .. name)
        refreshSaves()
    else
        setStatus("save failed")
    end
end

-- drag helper
local function makeDrag(handle, target)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos = true, i.Position, target.Position
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
makeDrag(header, main)
makeDrag(icon, icon)

btnBuild.MouseButton1Click:Connect(function()
    if running then return end
    local prompt = box.Text:gsub("^%s+", ""):gsub("%s+$", "")
    if prompt == "" then return setStatus("type a prompt") end

    local direct = prompt:match("^[Ll]oad%s+(.+)$")
    if direct then
        local name = safeName(direct)
        if fsOK() and isfile(CFG.saveDir .. "/" .. name .. ".json") then
            local ok, data = pcall(readfile, CFG.saveDir .. "/" .. name .. ".json")
            if ok and data then
                local ok2, raw = pcall(HttpService.JSONDecode, HttpService, data)
                if ok2 and type(raw) == "table" then
                    local plan = cleanPlan(raw)
                    if plan.count > 0 then return runPlan(plan, raw.prompt) end
                end
            end
        end
    end

    logLbl.Text = "asking model..."
    setStatus("thinking")
    task.spawn(function()
        local plan, err = askAI(prompt)
        if not plan then
            setStatus("ai error")
            logLbl.Text = tostring(err)
            return
        end
        local clean = cleanPlan(plan)
        if clean.count == 0 then
            setStatus("empty plan")
            logLbl.Text = HttpService:JSONEncode(plan):sub(1, 260)
            return
        end
        logLbl.Text = clean.name .. " · " .. #clean.walls .. "w " .. #clean.caps .. "c"
        build(clean, prompt)
    end)
end)
